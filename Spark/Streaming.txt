from pyspark.sql.functions import split, col
raw_stream = spark.readStream.format("socket") \
    .option("host", "localhost") \
    .option("port", 9999) \
    .load()
parsed_df = raw_stream.select(
    split(col("value"), ",").getItem(0).alias("sensor_id"),
    split(col("value"), ",").getItem(1).cast("int").alias("temperature")
)
alerts_df = parsed_df.filter(col("temperature") < 22)
query = alerts_df.writeStream \
    .outputMode("append") \
    .format("console") \
    .start()
